cmake_minimum_required(VERSION 3.16)
project(GrendelCppProjects LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(MSVC)
  add_compile_options(/W4)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

function(add_grendel_project target directory)
  add_executable(${target} "${directory}/main.cpp")
endfunction()

add_grendel_project(banking_system Banking_System)
add_grendel_project(cgpa_calculator CGPA_Calculator)
add_grendel_project(casino_number_guessing Casino_Number_Guessing_Game)
add_grendel_project(hospital_management Hospital_Management_System)
add_grendel_project(inventory_management Inventory_Management)
add_grendel_project(login_registration Login_and_Registration)
add_grendel_project(medical_information Medical_Information_System)
add_grendel_project(payroll_management Payroll_Management_System)
add_grendel_project(rock_paper_scissors Rock_Paper_Scissor)
add_grendel_project(scientific_calculator Scientific_Calculator)
add_grendel_project(store_management Store_Management_System)
add_grendel_project(student_data_management Student_Data_Management)
